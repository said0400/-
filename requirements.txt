# Python: 3.10+ required (tested on 3.10, 3.11, 3.12)
#
# ⚠️  Installation order in GitHub Actions:
#
#   1. System dependencies (Ubuntu):
#      sudo apt-get update
#      sudo apt-get install -y ffmpeg libgomp1 libsndfile1
#
#   2. Upgrade pip + setuptools:
#      pip install --upgrade pip setuptools wheel
#
#   3. Install PyTorch (CPU version):
#      pip install torch==2.5.1 torchaudio==2.5.1 \
#         --index-url https://download.pytorch.org/whl/cpu
#
#   4. Install openai-whisper (dependency for stable-ts):
#      pip install "setuptools<81.0.0" wheel
#      pip install openai-whisper==20240930 --no-build-isolation
#      pip install --upgrade setuptools
#
#   5. Install Python dependencies:
#      pip install -r requirements.txt
#
#   6. Install Node.js dependencies:
#      cd remotion && npm ci && cd ..
#
#   7. Install Playwright browsers (BOTH Python + Node.js):
#      python -m playwright install chromium --with-deps
#      cd remotion && npx playwright install chromium --with-deps && cd ..
#
#   ⚠️  NOTE: Steps 6+7 are required because render.mjs uses
#       Node.js Playwright (separate binary from Python Playwright).
#       Both must be installed independently.
#
# ══════════════════════════════════════════════════════════════════════
